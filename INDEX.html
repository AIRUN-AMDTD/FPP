<script>

  // Reemplaza con tu TikTok si quieres

  const TIKTOK_URL = "https://www.tiktok.com/@";

  const passthroughParams = ["utm_source","utm_medium","utm_campaign","ttclid","tt_ref","ref"];


  const fmt = (d)=> d ? new Date(d.getTime()-d.getTimezoneOffset()*60000).toISOString().slice(0,10) : "—";

  const parseDate = (s)=> { if(!s) return null; const d = new Date(s+"T12:00:00"); return isNaN(d)?null:d; };

  const addDays = (d, n)=> { const r = new Date(d); r.setDate(r.getDate()+n); return r; };

  const diffDays = (a,b)=> Math.round((a - b)/86400000);

  const qp = new URLSearchParams(location.search);


  // URL compartida sense dades (privacitat)

  function buildShareURL(){

    // No afegim inputs a l’URL; només mantenim UTM/ttclid si ja hi són

    const p = new URLSearchParams(location.search);

    for(const k of passthroughParams){ if(qp.has(k) && !p.has(k)) p.set(k, qp.get(k)); }

    return location.origin + location.pathname + (p.toString()?("?"+p.toString()):"");

  }


  function fromFUR(furStr, cycleLen){

    const fur = parseDate(furStr);

    if(!fur) return {error:"Introduce una FUR válida."};

    const adj = (Number(cycleLen)||28) - 28;

    const edd = addDays(fur, 280 + adj);

    const today = new Date();

    const gaDays = Math.max(0, Math.min(294, diffDays(today, fur)));

    const w = Math.floor(gaDays/7), d = gaDays%7;

    return {edd, gaDays, w, d, method:"FUR"};

  }


  function fromUltrasound(scanStr, w, d){

    const scan = parseDate(scanStr);

    if(!scan) return {error:"Introduce una fecha de eco válida."};

    const gaDays = (Number(w)||0)*7 + (Number(d)||0);

    if(gaDays<=0) return {error:"Introduce semanas/días de gestación válidos."};

    const remaining = 280 - gaDays;

    const edd = addDays(scan, remaining);

    const today = new Date();

    const lmpGuess = addDays(scan, -gaDays);

    const gaDaysToday = Math.max(0, Math.min(294, diffDays(today, lmpGuess)));

    const ww = Math.floor(gaDaysToday/7), dd = gaDaysToday%7;

    return {edd, gaDays:gaDaysToday, w:ww, d:dd, method:"Eco"};

  }


  function renderOut(el, data){

    if(data.error){ el.style.display="block"; el.innerHTML = `<span style="color:var(--err)">${data.error}</span>`; return; }

    const eddStr = fmt(data.edd);

    el.style.display="block";

    el.innerHTML = `

      <div><b>FPP:</b> ${eddStr}</div>

      <div><b>Semanas y días hoy:</b> ${data.w} semanas y ${data.d} días</div>

      <div class="muted">Método: ${data.method}</div>

    `;

    const kpi = document.getElementById('kpi');

    kpi.style.display='grid';

    document.getElementById('kpi_edd').textContent = eddStr;

    document.getElementById('kpi_ga').textContent = `${data.w} + ${data.d}`;

    window._last = {edd: eddStr, ga:`${data.w} semanas y ${data.d} días`, method:data.method};

  }

  document.getElementById('calcFUR').addEventListener('click', ()=>{

    const fur = document.getElementById('fur').value;

    const cycle = document.getElementById('cycle').value;

    const data = fromFUR(fur, cycle);

  });

    const gw = document.getElementById('ga_w').value;

    renderOut(document.getElementById('outB'), data);

  });

    const url = buildShareURL();

    try{

      await navigator.share({title:'Calculadora FPP', text:'Mi FPP', url});

    }catch(e){

      try{ await navigator.clipboard.writeText(url); alert('Enlace copiado.'); }catch(_){}

    }
